# 图片预览功能通过点击图片和背景退出全屏预览

## 产品经理

## 功能列表

1.  **图片全屏预览组件**
    *   在显示 `OrderItem.image_url` 的位置（如订单详情页、订单创建页的预览区域），将图片作为可交互元素。
    *   用户点击图片后，触发全屏或模态框形式的预览。
    *   预览层包含清晰的图片展示。
    *   点击预览层的背景（遮罩）区域，关闭预览，返回原页面。

## 验收标准

1.  **触发与展示**
    *   **Given** 用户位于订单详情页或订单创建/编辑页，且某个订单项包含有效的 `image_url`。
    *   **When** 用户点击该订单项中的图片缩略图。
    *   **Then** 页面应弹出全屏或模态框，清晰展示该图片的完整版本。

2.  **交互与退出**
    *   **Given** 图片预览层已打开。
    *   **When** 用户点击图片预览层中图片区域**之外**的任意背景（遮罩）位置。
    *   **Then** 预览层应关闭，用户返回至触发预览前的原页面状态。

3.  **边界情况**
    *   **Given** 订单项的 `image_url` 为 `null` 或无效。
    *   **Then** 不应渲染可点击的图片元素，或点击后无响应。
    *   **Given** 图片因加载失败无法显示。
    *   **Then** 预览层应显示明确的错误提示（如“图片加载失败”），并保持可通过点击背景关闭。

### 四、关键实现要点

1. **组件通信**

   - 通过props和emit实现父子组件通信

2. **边界情况处理**
   - 图片URL为null时不渲染可点击元素
   - 图片加载失败显示错误提示

   - 点击遮罩层关闭预览

   - 组件按需加载（使用defineAsyncComponent）
   - 图片懒加载（可选优化）

3. **兼容性测试**
   - 不同屏幕尺寸下的显示效果
   - 移动端触摸交互

<script setup lang="ts">
import { ref, watch, onMounted, onUnmounted } from 'vue'

const props = withDefaults(defineProps<{
  visible: boolean
  imageUrl: string | null
  imageName?: string | null
}>(), {
  visible: false,
  imageUrl: null,
  imageName: null
})

const emit = defineEmits<{
  (e: 'close'): void
}>()

const loadError = ref(false)
const loading = ref(false)

// 监听图片URL变化，重置状态
watch(() => props.imageUrl, (newUrl) => {
  if (newUrl) {
    loadError.value = false
    loading.value = true
  } else {
    loadError.value = false
    loading.value = false
  }
})

// 监听visible变化，处理键盘事件
watch(() => props.visible, (newVisible) => {
  if (newVisible) {
    // 添加键盘事件监听
    document.addEventListener('keydown', handleKeydown)
    // 禁止背景滚动
    document.body.style.overflow = 'hidden'
  } else {
    // 移除键盘事件监听
    document.removeEventListener('keydown', handleKeydown)
    // 恢复背景滚动
    document.body.style.overflow = ''
  }
})

onMounted(() => {
  // 如果初始就显示，也需要处理
  if (props.visible) {
    document.addEventListener('keydown', handleKeydown)
    document.body.style.overflow = 'hidden'
  }
})

onUnmounted(() => {
  document.removeEventListener('keydown', handleKeydown)
  document.body.style.overflow = ''
})

// 关闭预览
function close() {
  emit('close')
}

// 处理键盘事件
function handleKeydown(event: KeyboardEvent) {
  if (event.key === 'Escape') {
    close()
  }
}

// 图片加载错误处理
function handleError() {
  loadError.value = true
  loading.value = false
}

// 图片加载成功处理
function handleLoad() {
  loading.value = false
}

// 重新加载图片
function retryLoad() {
  if (props.imageUrl) {
    loadError.value = false
    loading.value = true
    // 通过添加时间戳强制浏览器重新加载
    const img = new Image()
    img.src = props.imageUrl + '?t=' + Date.now()
    img.onload = () => {
      loading.value = false
    }
    img.onerror = () => {
      loadError.value = true
      loading.value = false
    }
  }
}
</script>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useRoute } from 'vue-router'
import { useOrderStore } from '../stores/order'
import ImageViewer from '../components/ImageViewer.vue'
import type { Order } from '../types'

const route = useRoute()
const orderStore = useOrderStore()

const order = ref<Order | null>(null)
const loading = ref(false)

// 图片预览相关状态
const showImageViewer = ref(false)
const currentPreviewImage = ref<string | null>(null)
const currentImageName = ref<string | null>(null)

// 打开图片预览
function openImageViewer(imageUrl: string | null, imageName?: string) {
  if (imageUrl) {
    currentPreviewImage.value = imageUrl
    currentImageName.value = imageName || null
    showImageViewer.value = true
  }
}

// 关闭图片预览
function closeImageViewer() {
  showImageViewer.value = false
  // 延迟清空图片URL，避免关闭动画时图片消失
  setTimeout(() => {
    currentPreviewImage.value = null
    currentImageName.value = null
  }, 300)
}

// 获取订单详情
async function fetchOrderDetail() {
  const orderId = route.params.id as string
  if (!orderId) return
  
  loading.value = true
  try {
    const data = await orderStore.fetchOrderById(orderId)
    order.value = data
  } catch (error) {
    console.error('获取订单详情失败:', error)
  } finally {
    loading.value = false
  }
}

onMounted(() => {
  fetchOrderDetail()
})
</script>
