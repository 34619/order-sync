# cnc加工和3d打印完成后自动下发车间任务，车间完成后才能交付

## 产品经理

## 功能列表

1. **自动任务分发**
   - 当CNC加工任务状态变为"completed"时，自动为该订单创建车间任务
   - 当3D打印任务状态变为"completed"时，自动为该订单创建车间任务

2. **交付状态控制**
   - 车间任务状态为"completed"时，订单才能变更为"pending_delivery"状态
   - 订单状态变更时验证前置条件

3. **任务关联管理**
   - 车间任务自动关联原始订单
   - 任务创建时设置合理的默认值

### 技术方案：自动任务分发与交付状态控制

#### 一、 需修改的文件
1.  **数据库 Schema (`supabase/schema.sql`)**：
    *   为 `order_tasks` 表添加 `task_type` 字段（用于区分任务归属部门）。
    *   创建触发器函数 `handle_task_completion()` 和对应的触发器。
    *   创建函数 `check_order_ready_for_delivery()` 用于状态校验。

2.  **服务端 API (`server/src/index.ts`)**：
    *   修改 `PUT /api/tasks/:id` 端点，在任务状态更新为 `completed` 后，调用自动创建车间任务的逻辑。
    *   修改 `PUT /api/orders/:id/status` 端点，添加在更新为 `pending_delivery` 前的校验逻辑。

3.  **客户端类型 (`client/src/types/index.ts`)**：
    *   为 `OrderTask` 接口添加 `task_type: Department` 字段。

#### 二、 数据变更
1.  **`order_tasks` 表新增字段**：
    *   `task_type` (类型: `text`, 约束: `check (task_type in ('cnc_program', 'cnc_machine', 'print_3d', 'workshop'))`, 非空)。用于明确标识任务属于哪个生产环节。

2.  **新增数据库函数**：
    *   `handle_task_completion()`: 当任务状态更新为 `completed` 时，根据 `task_type` 判断是否需要为同一订单自动创建 `workshop` 类型的车间任务。
    *   `check_order_ready_for_delivery(order_id uuid)`: 检查指定订单下是否存在未完成 (`status != 'completed'`) 的车间任务 (`task_type = 'workshop'`)。返回 `boolean`。

#### 三、 实现步骤
**第一步：扩展数据库**
1.  在 `supabase/schema.sql` 的 `order_tasks` 表定义中，添加 `task_type text not null` 字段及相应约束。
2.  编写并创建触发器函数 `handle_task_completion()`。逻辑示例：
    ```sql
    create or replace function handle_task_completion()
    returns trigger as $$
    begin
      -- 当任务状态变为‘completed’，且任务类型为需要触发后续流程的类型时
      if NEW.status = 'completed' and OLD.status != 'completed' and NEW.task_type in ('cnc_program', 'cnc_machine', 'print_3d') then
        -- 检查该订单是否已存在未完成的车间任务，避免重复创建
        if not exists (
          select 1 from order_tasks
          where order_id = NEW.order_id
            and task_type = 'workshop'
            and status != 'completed'
        ) then
          -- 为订单创建一个新的车间任务
          insert into order_tasks (order_id, task_type, status, created_at)
          values (NEW.order_id, 'workshop', 'pending', now());
        end if;
      end if;
      return NEW;
    end;
    $$ language plpgsql security definer;
    ```
3.  在 `order_tasks` 表上创建 `after update` 触发器，调用上述函数。
4.  编写并创建函数 `check_order_ready_for_delivery(order_id uuid)`。逻辑示例：
    ```sql
    create or replace function check_order_ready_for_delivery(order_id uuid)
    returns boolean as $$
    begin
      return not exists (
        select 1 from order_tasks
        where order_id = $1
          and task_type = 'workshop'
          and status != 'completed'
      );
    end;
    $$ language plpgsql stable;
    ```

**第二步：修改服务端 API**
1.  **修改任务更新端点 (`PUT /api/tasks/:id`)**:
    *   在成功更新任务状态后，检查新状态是否为 `completed`。
    *   （可选，若使用数据库触发器则此步可省略）若使用应用层逻辑，则在此处根据更新后任务的 `task_type` 和 `order_id`，查询是否需要创建车间任务，并执行插入操作。**推荐使用数据库触发器以保证数据一致性。**
2.  **修改订单状态更新端点 (`PUT /api/orders/:id/status`)**:
    *   当请求将状态更新为 `pending_delivery` 时，首先调用数据库函数 `check_order_ready_for_delivery(order_id)` 进行校验。
    *   若函数返回 `false`（即存在未完成的车间任务），则返回 `400` 错误及明确提示（如：“订单尚有车间任务未完成，无法进入待交付状态”）。
    *   若校验通过，则允许更新订单状态。

**第三步：更新客户端类型与UI**
1.  在 `client/src/types/index.ts` 的 `OrderTask` 接口中，添加 `task_type: Department` 字段。
2.  在订单详情页 (`OrderDetail.vue`) 中，确保列表渲染了任务的 `task_type`，使用户能清晰看到自动创建的车间任务。
3.  在订单状态变更的操作处（如下拉菜单或按钮），捕获后端返回的校验错误，并以友好提示（如Toast或Alert）展示给用户。

create or replace function generate_order_number(customer_code text)
returns text as $$
declare
  year_month text;
  sequence_val int;
  new_order_number text;
begin
  -- 获取当前年月，格式：202401
  year_month := to_char(now(), 'YYYYMM');
  
  -- 获取当月订单序号
  select coalesce(max(
    cast(substring(order_number from length(customer_code) + year_month::text + 2) as int)
  ), 0) + 1 into sequence_val
  from orders
  where order_number like customer_code || year_month || '%';
  
  -- 生成订单号：客户代码 + 年月 + 4位序号
  new_order_number := customer_code || year_month || lpad(sequence_val::text, 4, '0');
  
  return new_order_number;
end;
$$ language plpgsql;

-- 函数：处理任务完成（核心：自动创建车间任务）
-- =============================================
create or replace function handle_task_completion()
returns trigger as $$
begin
  -- 当任务状态变为'completed'时
  if NEW.status = 'completed' and OLD.status != 'completed' then
    -- 检查任务类型，如果是CNC编程、CNC加工、3D打印之一
    if NEW.task_type in ('cnc_program', 'cnc_machine', 'print_3d') then
      -- 检查该订单是否已存在未完成的车间任务
      if not exists (
        select 1 from order_tasks
        where order_id = NEW.order_id
          and task_type = 'workshop'
          and status != 'completed'
      ) then
        -- 为订单创建一个新的车间任务
        insert into order_tasks (order_id, task_type, status, created_at, updated_at)
        values (NEW.order_id, 'workshop', 'pending', now(), now());
        
        -- 记录操作日志
        insert into operation_logs (table_name, record_id, operation, new_data, performed_by)
        values (
          'order_tasks', 
          NEW.id, 
          'update',
          jsonb_build_object(
            'auto_created', true,
            'task_type', 'workshop',
            'order_id', NEW.order_id,
            'created_for', NEW.task_type
          ),
          auth.uid()
        );
      end if;
    end if;
  end if;
  
  return NEW;
end;
$$ language plpgsql security definer;

-- =============================================
-- 触发器：任务完成时自动创建车间任务
-- =============================================
create trigger task_completion_trigger
  after update on order_tasks
  for each row execute function handle_task_completion();

-- =============================================
-- 函数：检查订单是否可以进入待交付状态
-- =============================================
create or replace function check_order_ready_for_delivery(order_id uuid)
returns boolean as $$
begin
  -- 检查是否存在未完成的车间任务
  return not exists (
    select 1 from order_tasks
    where order_id = $1
      and task_type = 'workshop'
      and status != 'completed'
  );
end;
$$ language plpgsql stable;

-- =============================================
-- 函数：更新订单状态（带校验）
-- =============================================
create or replace function update_order_status(order_id uuid, new_status text)
returns boolean as $$
declare
  current_status text;
  can_update boolean := false;
begin
  -- 获取当前状态
  select status into current_status from orders where id = order_id;
  
  -- 状态流转逻辑
  case current_status
    when 'pending_dispatch' then
      can_update := new_status in ('programming', 'rough_machining', 'fine_machining');
    when 'programming' then
      can_update := new_status in ('rough_machining', 'fine_machining');
    when 'rough_machining' then
      can_update := new_status in ('fine_machining');
    when 'fine_machining' then
      -- 进入待交付状态前，检查车间任务是否完成
      if new_status = 'pending_delivery' then
        can_update := check_order_ready_for_delivery(order_id);
      else
        can_update := false;
      end if;
    when 'pending_delivery' then
      can_update := new_status = 'delivered';
    when 'delivered' then
      can_update := false; -- 已交付状态不能更改
    else
      can_update := false;
  end case;
  
  -- 如果允许更新，执行更新
  if can_update then
    update orders 
    set status = new_status, updated_at = now()
    where id = order_id;
    
    -- 记录状态变更日志
    insert into operation_logs (table_name, record_id, operation, old_data, new_data, performed_by)
    values (
      'orders',
      order_id,
      'update',
      jsonb_build_object('status', current_status),
      jsonb_build_object('status', new_status),
      auth.uid()
    );
    
    return true;
  end if;
  
  return false;
end;
$$ language plpgsql security definer;

// 验证 JWT 并获取用户信息
async function authenticateUser(req: express.Request, res: express.Response, next: express.NextFunction) {
  const authHeader = req.headers.authorization
  if (!authHeader?.startsWith('Bearer ')) {
    return res.status(401).json({ error: '未授权' })
  }

  const token = authHeader.slice(7)
  const { data: { user }, error } = await supabase.auth.getUser(token)

  if (error || !user) {
    return res.status(401).json({ error: '无效的令牌' })
  }

  const { data: profile } = await supabase
    .from('profiles')
    .select('id, role, username')
    .eq('id', user.id)
    .single()

  if (!profile) {
    return res.status(401).json({ error: '用户档案不存在' })
  }

  ;(req as any).userId = user.id
  ;(req as any).userRole = profile.role
  ;(req as any).username = profile.username
  next()
}

// 验证管理员权限
async function requireAdmin(req: express.Request, res: express.Response, next: express.NextFunction) {
  if ((req as any).userRole !== 'admin') {
    return res.status(403).json({ error: '需要管理员权限' })
  }
  next()
}

// 验证办公室或管理员权限
async function requireOfficeOrAdmin(req: express.Request, res: express.Response, next: express.NextFunction) {
  const role = (req as any).userRole
  if (role !== 'admin' && role !== 'office') {
    return res.status(403).json({ error: '需要办公室或管理员权限' })
  }
  next()
}
