### 后端上传接口
```typescript
// server/src/routes/upload.ts
app.post('/api/upload/image', requireAuth, async (req, res) => {
  // 1. 验证角色
  if (!['admin', 'office'].includes(req.userRole)) {
    return res.status(403).json({ error: '无上传权限' })
  }
  
  // 2. 验证文件
  const file = req.file
  if (!file) return res.status(400).json({ error: '未提供文件' })
  
  const allowedTypes = ['image/jpeg', 'image/png']
  if (!allowedTypes.includes(file.mimetype)) {
    return res.status(400).json({ error: '仅支持JPG/PNG格式' })
  }
  
  if (file.size > 2 * 1024 * 1024) {
    return res.status(400).json({ error: '文件大小不能超过2MB' })
  }
  
  // 3. 上传到Supabase Storage
  const fileName = `${Date.now()}_${file.originalname}`
  const { data, error } = await supabase.storage
    .from('product-images')
    .upload(fileName, file.buffer)
  
  if (error) throw error
  
  // 4. 返回URL
  const url = `/api/images/${fileName}`
  res.json({ url })
})
```

// 验证JWT并检查权限（通用版本）
async function requireAuth(req: express.Request, res: express.Response, next: express.NextFunction) {
  const authHeader = req.headers.authorization
  if (!authHeader?.startsWith('Bearer ')) {
    return res.status(401).json({ error: '未授权' })
  }

  const token = authHeader.slice(7)
  const { data: { user }, error } = await supabase.auth.getUser(token)

  if (error || !user) {
    return res.status(401).json({ error: '无效的令牌' })
  }

  const { data: profile } = await supabase
    .from('profiles')
    .select('role')
    .eq('id', user.id)
    .single()

  ;(req as any).userId = user.id
  ;(req as any).userRole = profile?.role
  next()
}

// 验证管理员权限
async function requireAdmin(req: express.Request, res: express.Response, next: express.NextFunction) {
  await requireAuth(req, res, () => {
    if ((req as any).userRole !== 'admin') {
      return res.status(403).json({ error: '需要管理员权限' })

      return res.status(403).json({ error: '无上传权限' })

app.get('/api/images/*', requireAuth, async (req, res) => {
  try {
    const path = req.params[0]
    if (!path) {
      return res.status(400).json({ error: '缺少路径参数' })
    }

    // 从Supabase Storage获取文件
    const { data, error } = await supabase.storage
      .from('product-images')
      .download(path)

    if (error) {
      console.error('下载文件失败:', error)
      return res.status(404).json({ error: '文件不存在' })
    }

    // 设置响应头
    const ext = path.split('.').pop()?.toLowerCase()
    let contentType = 'image/jpeg'
    if (ext === 'png') contentType = 'image/png'
    if (ext === 'webp') contentType = 'image/webp'

    res.set({
      'Content-Type': contentType,
      'Cache-Control': 'public, max-age=31536000', // 缓存1年
      'Content-Disposition': 'inline'
    })

    // 将ReadableStream转换为Buffer
    const chunks: Uint8Array[] = []
    const reader = data.getReader()
    
    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      chunks.push(value)
    }
    
    const buffer = Buffer.concat(chunks)
    res.send(buffer)
  } catch (error) {
    console.error('代理图片错误:', error)
    res.status(500).json({ error: '服务器内部错误' })
  }
})

## 审查结果

### 严重问题

1. **权限中间件逻辑错误**（server/src/index.ts）
   - `requireAdmin` 和 `requireUploadPermission` 中间件调用 `requireAuth` 的方式不正确
   - 当 `requireAuth` 调用 `next()` 时，内层回调会立即执行，但外层中间件可能还未完成
   - 应改为在 `requireAuth` 成功后调用权限检查逻辑

2. **数据库表删除风险**（supabase/schema.sql）
   - `drop table if exists order_items cascade;` 会删除已有的 `order_items` 表
   - 如果这是更新脚本，应该使用 `ALTER TABLE` 添加字段而不是删除重建

### 安全问题

1. **存储桶策略不完整**
   - 只有插入和选择策略，缺少更新和删除策略
   - 建议添加：
   ```sql
   create policy "Authenticated users can update images"
   on storage.objects for update
   to authenticated
   using (bucket_id = 'product-images');
   
   create policy "Authenticated users can delete images"
   on storage.objects for delete
   to authenticated
   using (bucket_id = 'product-images');
   ```

2. **路径遍历风险**
   - 图片访问代理接口没有验证路径参数，可能导致路径遍历攻击
   - 应该对路径进行规范化验证

3. **重复的类型验证**
   - 文件类型在 multer 的 `fileFilter` 和路由处理器中都进行了验证，存在冗余

### 性能/可靠性问题

1. **错误处理不完整**
   - 错误处理中间件只处理了 multer 错误，其他错误可能未被捕获
   - 建议统一错误处理

2. **流处理可能不兼容**
   - `data.getReader()` 假设返回的是 ReadableStream，但 Supabase 的 `download` 方法可能返回 Blob
   - 应该检查返回类型或使用 `arrayBuffer()` 方法

3. **订单更新没有事务支持**
   - 更新订单时先删除所有 items 再插入新的，如果插入失败会导致数据丢失
   - 建议使用事务或检查插入成功后再删除
